```sql
-- supabase/migrations/236_workshop_items.sql
-- Workshop items: task-scoped working assets (replaces folders.name='temp' convention).
-- See docs/superpowers/specs/2026-05-28-workshop-design.md § Schema.

CREATE TABLE IF NOT EXISTS public.workshop_items (
    id            BIGINT PRIMARY KEY DEFAULT generate_snowflake_id(),

    task_type     TEXT NOT NULL
                  CHECK (task_type IN ('issue', 'chat_session')),
    task_id       BIGINT NOT NULL,

    resource_id   BIGINT NOT NULL
                  REFERENCES public.resources(id) ON DELETE CASCADE,

    team_id       BIGINT NOT NULL
                  REFERENCES public.teams(id) ON DELETE CASCADE,
    project_id    BIGINT,

    created_by    UUID NOT NULL REFERENCES auth.users(id),
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    is_trashed    BOOLEAN NOT NULL DEFAULT FALSE,
    trashed_at    TIMESTAMPTZ
);

CREATE INDEX IF NOT EXISTS idx_workshop_task
    ON public.workshop_items (task_type, task_id)
    WHERE is_trashed = false;

CREATE INDEX IF NOT EXISTS idx_workshop_project
    ON public.workshop_items (project_id)
    WHERE project_id IS NOT NULL AND is_trashed = false;

CREATE INDEX IF NOT EXISTS idx_workshop_team
    ON public.workshop_items (team_id)
    WHERE is_trashed = false;

CREATE INDEX IF NOT EXISTS idx_workshop_resource
    ON public.workshop_items (resource_id)
    WHERE is_trashed = false;

-- RLS: team_members of the owning team can read/write
ALTER TABLE public.workshop_items ENABLE ROW LEVEL SECURITY;

CREATE POLICY workshop_items_select ON public.workshop_items
    FOR SELECT
    USING (
        team_id IN (
            SELECT team_id FROM public.team_members WHERE user_id = auth.uid()
        )
    );

CREATE POLICY workshop_items_insert ON public.workshop_items
    FOR INSERT
    WITH CHECK (
        team_id IN (
            SELECT team_id FROM public.team_members WHERE user_id = auth.uid()
        )
    );

CREATE POLICY workshop_items_update ON public.workshop_items
    FOR UPDATE
    USING (
        team_id IN (
            SELECT team_id FROM public.team_members WHERE user_id = auth.uid()
        )
    );

CREATE POLICY workshop_items_delete ON public.workshop_items
    FOR DELETE
    USING (
        team_id IN (
            SELECT team_id FROM public.team_members WHERE user_id = auth.uid()
        )
    );

NOTIFY pgrst, 'reload schema';
```

```python
# backend/app/repositories/workshop_repository.py
"""WorkshopRepository — CRUD for workshop_items.

Workshop items are task-scoped working assets. See
docs/superpowers/specs/2026-05-28-workshop-design.md.
"""

from __future__ import annotations

from typing import Any, Optional


class WorkshopRepository:
    """Direct asyncpg CRUD (matches resources_repository pattern)."""

    async def create_item(
        self,
        *,
        task_type: str,
        task_id: int,
        resource_id: int,
        team_id: int,
        project_id: Optional[int],
        created_by: str,
    ) -> str:
        """Insert a new workshop_items row. Returns the new id as string."""
        from app.db import engine as db_engine

        rows = await db_engine.fetch_all(
            """
            INSERT INTO public.workshop_items
                (task_type, task_id, resource_id, team_id, project_id, created_by)
            VALUES
                (:task_type, :task_id, :resource_id, :team_id, :project_id, :created_by)
            RETURNING id::text
            """,
            {
                "task_type": task_type,
                "task_id": task_id,
                "resource_id": resource_id,
                "team_id": team_id,
                "project_id": project_id,
                "created_by": created_by,
            },
        )
        return rows[0]["id"]

    async def get_item(self, *, workshop_item_id: int) -> Optional[dict]:
        """Fetch one item joined with resources for file metadata."""
        from app.db import engine as db_engine

        rows = await db_engine.fetch_all(
            """
            SELECT
                wi.id::text, wi.task_type, wi.task_id::text, wi.resource_id::text,
                wi.team_id::text, wi.project_id::text, wi.created_by::text,
                wi.created_at, wi.is_trashed,
                r.filename, r.mime_type, r.file_size, r.file_path
              FROM public.workshop_items wi
              JOIN public.resources r ON r.id = wi.resource_id
             WHERE wi.id = :id AND wi.is_trashed = false
             LIMIT 1
            """,
            {"id": workshop_item_id},
        )
        return rows[0] if rows else None

    async def list_by_task(
        self,
        *,
        task_type: str,
        task_id: int,
    ) -> list[dict]:
        """List non-trashed workshop items for one task."""
        from app.db import engine as db_engine

        rows = await db_engine.fetch_all(
            """
            SELECT
                wi.id::text, wi.task_type, wi.task_id::text,
                wi.resource_id::text, wi.team_id::text, wi.project_id::text,
                wi.created_by::text, wi.created_at,
                r.filename, r.mime_type, r.file_size
              FROM public.workshop_items wi
              JOIN public.resources r ON r.id = wi.resource_id
             WHERE wi.task_type = :task_type
               AND wi.task_id = :task_id
               AND wi.is_trashed = false
             ORDER BY wi.created_at DESC
            """,
            {"task_type": task_type, "task_id": task_id},
        )
        return rows or []

    async def list_by_team(
        self,
        *,
        team_id: int,
        limit: int = 50,
    ) -> list[dict]:
        """List a team's recent workshop items (sidebar Workshop view)."""
        from app.db import engine as db_engine

        rows = await db_engine.fetch_all(
            """
            SELECT
                wi.id::text, wi.task_type, wi.task_id::text,
                wi.resource_id::text, wi.team_id::text, wi.project_id::text,
                wi.created_by::text, wi.created_at,
                r.filename, r.mime_type, r.file_size
              FROM public.workshop_items wi
              JOIN public.resources r ON r.id = wi.resource_id
             WHERE wi.team_id = :team_id
               AND wi.is_trashed = false
             ORDER BY wi.created_at DESC
             LIMIT :limit
            """,
            {"team_id": team_id, "limit": min(max(int(limit), 1), 200)},
        )
        return rows or []

    async def soft_delete(self, *, workshop_item_id: int) -> None:
        """Mark item as trashed."""
        from app.db import engine as db_engine

        await db_engine.fetch_all(
            """
            UPDATE public.workshop_items
               SET is_trashed = true, trashed_at = NOW()
             WHERE id = :id
            """,
            {"id": workshop_item_id},
        )

    async def hard_delete(self, *, workshop_item_id: int) -> None:
        """Permanently delete (used by promote flow — graduate to library)."""
        from app.db import engine as db_engine

        await db_engine.fetch_all(
            "DELETE FROM public.workshop_items WHERE id = :id",
            {"id": workshop_item_id},
        )
```

```python
# backend/app/services/library/workshop_promote.py
"""Promote a workshop item to a permanent resource_items row.

After promote:
  - resource_items row exists with the target library / folder
  - workshop_items row is hard-deleted (intentional graduation, not trash)
  - underlying resources row is unchanged
"""

from __future__ import annotations

from typing import Optional

from loguru import logger

from app.repositories.resources_repository import ResourcesRepository
from app.repositories.workshop_repository import WorkshopRepository


async def promote_workshop_item(
    *,
    workshop_item_id: int,
    target_folder_id: Optional[int],
    target_library_id: Optional[int],
) -> dict:
    """Move resource from workshop to library. Returns the new resource_items row."""
    workshop_repo = WorkshopRepository()
    resources_repo = ResourcesRepository()

    item = await workshop_repo.get_item(workshop_item_id=workshop_item_id)
    if item is None:
        raise ValueError(f"workshop item {workshop_item_id} not found")

    # Default library: the team's first library if not specified.
    library_id = target_library_id
    if library_id is None:
        # Personal team has implicit default library — handled by resources_repo
        # selecting team's first library by sort_order. For workshop promote
        # without explicit target, this is the "save to library root" path.
        library_id = await _resolve_default_library(item["team_id"])

    insert_payload = {
        "resource_id": item["resource_id"],
        "scope_type": "team",  # post-Spec-1, always 'team'
        "scope_id": item["team_id"],
        "library_id": library_id,
        "folder_id": target_folder_id,
    }

    new_item = await resources_repo.create_resource_item(insert_payload)
    await workshop_repo.hard_delete(workshop_item_id=workshop_item_id)

    logger.info(
        f"[workshop_promote] item={workshop_item_id} → "
        f"resource_item={new_item.get('id')} library={library_id} folder={target_folder_id}"
    )
    return new_item


async def _resolve_default_library(team_id: int) -> Optional[int]:
    """Resolve the team's default library id. Returns None if team has none
    (e.g., personal team without explicit library — caller falls back to
    folder-less / library-less scope_items, which is valid)."""
    from app.db import engine as db_engine

    rows = await db_engine.fetch_all(
        "SELECT id FROM public.libraries WHERE scope_id = :tid::text "
        "ORDER BY sort_order LIMIT 1",
        {"tid": team_id},
    )
    return rows[0]["id"] if rows else None
```

```python
"""save_chat_session_upload — chat composer upload entry point.

Replaces the legacy app.services.library.chat_upload.save_chat_temp_upload.
Each call:
  1. Resolves session → team_id, project_id
  2. Saves file bytes to disk via resources_service
  3. Creates a workshop_items row binding the new resource to the session
"""

from __future__ import annotations

from typing import Optional

from loguru import logger

from app.repositories.workshop_repository import WorkshopRepository


async def save_chat_session_upload(
    *,
    user_id: str,
    session_id: int,
    file_bytes: bytes,
    filename: str,
    mime: str,
) -> dict:
    """Save file to disk + create resources row + create workshop_items row.

    Returns: {workshop_item_id, resource_id, filename, file_path}.
    """
    team_id, project_id = await _resolve_session_team_and_project(session_id)

    resource = await _save_file_and_resource(
        user_id=user_id,
        team_id=team_id,
        file_bytes=file_bytes,
        filename=filename,
        mime=mime,
    )

    workshop_id = await WorkshopRepository().create_item(
        task_type="chat_session",
        task_id=session_id,
        resource_id=int(resource["resource_id"]),
        team_id=team_id,
        project_id=project_id,
        created_by=user_id,
    )

    logger.info(
        f"[workshop_upload] saved {filename} ({len(file_bytes)}B) → "
        f"workshop_item={workshop_id} resource={resource['resource_id']} "
        f"team={team_id} session={session_id}"
    )
    return {
        "workshop_item_id": workshop_id,
        "resource_id": resource["resource_id"],
        "filename": resource["filename"],
        "file_path": resource["file_path"],
    }


async def _resolve_session_team_and_project(session_id: int) -> tuple[int, Optional[int]]:
    """Look up team_id and project_id for an ai_session."""
    from app.db import engine as db_engine

    rows = await db_engine.fetch_all(
        "SELECT team_id, project_id FROM public.ai_sessions WHERE id = :sid",
        {"sid": session_id},
    )

    return int(rows[0]["team_id"]), (int(rows[0]["project_id"]) if rows[0]["project_id"] else None)
```

```python
# backend/app/services/library/workshop_attach.py
"""Helper for AI tools / DBOS workflows to attach an existing resource to a task.

Use this when the resource row already exists (e.g., AI tool wrote a generated
image directly to storage and inserted the resources row itself, then needs to
bind it to the current chat session or issue).
"""

from __future__ import annotations

from typing import Literal, Optional

from loguru import logger

from app.repositories.workshop_repository import WorkshopRepository

TaskType = Literal["issue", "chat_session"]


async def attach_to_task(
    *,
    task_type: TaskType,
    task_id: int,
    resource_id: int,
    created_by: str,
) -> str:
    """Create a workshop_items row attaching an existing resource to a task.

    Derives team_id, project_id from the task. Returns the new workshop_items.id.
    """
    if task_type not in ("issue", "chat_session"):
        raise ValueError(
            f"task_type must be one of: 'issue', 'chat_session' (got {task_type!r})"
        )

    team_id, project_id = await _resolve_task_team_and_project(task_type, task_id)

    workshop_id = await WorkshopRepository().create_item(
        task_type=task_type,
        task_id=task_id,
        resource_id=resource_id,
        team_id=team_id,
        project_id=project_id,
        created_by=created_by,
    )
    logger.info(
        f"[workshop_attach] resource={resource_id} → workshop_item={workshop_id} "
        f"task={task_type}/{task_id} team={team_id}"
    )
    return workshop_id


async def _resolve_task_team_and_project(
    task_type: TaskType, task_id: int
) -> tuple[int, Optional[int]]:
    """Derive (team_id, project_id) from the task row."""
    from app.db import engine as db_engine

    if task_type == "chat_session":
        rows = await db_engine.fetch_all(
            "SELECT team_id, project_id FROM public.ai_sessions WHERE id = :id",
            {"id": task_id},
        )
    elif task_type == "issue":
        rows = await db_engine.fetch_all(
            "SELECT team_id, project_id FROM public.issues WHERE id = :id",
            {"id": task_id},
        )
    else:
        raise AssertionError("unreachable")

    if not rows:
        raise ValueError(f"{task_type} {task_id} not found")
    return int(rows[0]["team_id"]), (int(rows[0]["project_id"]) if rows[0]["project_id"] else None)
```

```python
from __future__ import annotations

from typing import Optional

from fastapi import APIRouter, Depends, HTTPException, Query

from app.core.deps import AuthDep
from app.repositories.workshop_repository import WorkshopRepository

router = APIRouter(prefix="/api/v1/workshop", tags=["workshop"])


@router.get("")
async def search_workshop(
    auth: AuthDep,
    task_type: Optional[str] = Query(None),
    task_id: Optional[int] = Query(None),
    team_id: Optional[int] = Query(None),
    project_id: Optional[int] = Query(None),
    limit: int = Query(50, ge=1, le=200),
):
    """Search workshop items by task / team / project."""
    repo = WorkshopRepository()
    if task_type and task_id:
        results = await repo.list_by_task(task_type=task_type, task_id=task_id)
    elif team_id:
        results = await repo.list_by_team(team_id=team_id, limit=limit)
    else:
        raise HTTPException(status_code=400, detail="provide task_type+task_id or team_id")
    return {"results": results, "count": len(results)}

    item = await WorkshopRepository().get_item(workshop_item_id=workshop_item_id)

        raise HTTPException(status_code=404, detail="not found")
    return item
```

```python
# backend/app/workflows/workshop_sweeper.py
"""DBOS scheduled workflow: TTL-sweep + orphan-cascade for workshop_items.

Daily cron tick. Two passes:
  1. TTL sweep: items older than the team's chat_temp_ttl_days setting → soft-delete
  2. Orphan sweep: items whose polymorphic task_id parent no longer exists → soft-delete
"""

from __future__ import annotations

from dbos import DBOS  # type: ignore[import-not-found]
from loguru import logger

from app.services.library.temp_ttl_settings import get_chat_temp_ttl_days


async def sweep_workshop_items() -> dict:
    """One-pass sweeper. Returns counts dict."""
    from app.db import engine as db_engine

    # TTL pass — query teams + their workshop_ttl_days settings
    ttl_rows = await db_engine.fetch_all(
        """
        UPDATE public.workshop_items wi
           SET is_trashed = true, trashed_at = NOW()
          FROM public.teams t
         WHERE wi.team_id = t.id
           AND wi.is_trashed = false
           AND wi.created_at < NOW() - (
               COALESCE((t.settings_json->>'workshop_ttl_days')::int, 7)
               * INTERVAL '1 day'
           )
         RETURNING wi.id
        """,
        {},
    )

    # Orphan pass
    orphan_rows = await db_engine.fetch_all(
        """
        UPDATE public.workshop_items
           SET is_trashed = true, trashed_at = NOW()
         WHERE is_trashed = false
           AND (
               (task_type = 'issue'
                AND task_id NOT IN (SELECT id FROM public.issues))
               OR
               (task_type = 'chat_session'
                AND task_id NOT IN (SELECT id FROM public.ai_sessions))
           )
         RETURNING id
        """,
        {},
    )

    ttl_count = len(ttl_rows or [])
    orphan_count = len(orphan_rows or [])
    logger.info(
        f"[workshop_sweeper] ttl_swept={ttl_count} orphan_swept={orphan_count}"
    )
    return {"ttl_swept": ttl_count, "orphan_swept": orphan_count}


@DBOS.scheduled("0 4 * * *")  # daily at 04:00 UTC
@DBOS.workflow()
async def workshop_sweeper_scheduled(*, scheduled_time, actual_time) -> dict:
    return await sweep_workshop_items()


__all__ = ["sweep_workshop_items", "workshop_sweeper_scheduled"]
```

```sql
-- scripts/migrations/2026-05-28-workshop-legacy-migrate.sql
-- Migrate legacy folders.name='temp' resources into workshop_items.
--
-- Heuristic: best-effort assignment to most recent ai_session by the same
-- creator within the same team. If no match, the resource is dropped from
-- workshop (the resource_items row remains as a permanent personal upload).
-- See spec § Data Migration.

BEGIN;

-- For each temp folder, for each resource in it, attempt to attach to
-- the most recent ai_session by the same creator in the same team.
INSERT INTO public.workshop_items
    (task_type, task_id, resource_id, team_id, project_id, created_by, created_at)
SELECT
    'chat_session',
    s.id,
    ri.resource_id,
    s.team_id,
    s.project_id,
    ri.added_by,
    ri.created_at
  FROM public.resource_items ri
  JOIN public.folders f ON ri.folder_id = f.id
  JOIN LATERAL (
      SELECT id, team_id, project_id
        FROM public.ai_sessions
       WHERE user_id = ri.added_by
         AND team_id::text = ri.scope_id
       ORDER BY created_at DESC
       LIMIT 1
  ) s ON true
 WHERE f.name = 'temp'
   AND f.is_trashed = false
   AND ri.is_trashed = false
   AND NOT EXISTS (
       SELECT 1 FROM public.workshop_items wi
        WHERE wi.resource_id = ri.resource_id
   );

-- For resources in temp folders that had no matching session, leave them
-- as-is (they remain visible as permanent uploads). Mark the temp folders
```

```typescript
import { getAuthHeaders } from './parserService';

const API = import.meta.env.VITE_API_URL || '';

export type WorkshopItem = {
  id: string;
  task_type: 'issue' | 'chat_session';
  task_id: string;
  resource_id: string;
  team_id: string;
  project_id: string | null;
  created_by: string;
  created_at: string;
  filename: string;
  mime_type: string | null;
  file_size: number | null;
};

export async function listByTeam(teamId: string, limit = 50): Promise<{results: WorkshopItem[], count: number}> {
  const headers = await getAuthHeaders();
  const u = new URL(`${API}/api/v1/workshop`, window.location.origin);
  u.searchParams.set('team_id', teamId);
  u.searchParams.set('limit', String(limit));
  const r = await fetch(u.toString(), { headers });
  if (!r.ok) throw new Error(`workshop list failed: ${r.status}`);
  return r.json();
}

export async function promote(workshopItemId: string, opts: {target_folder_id?: string, target_library_id?: string}): Promise<{resource_item_id: string}> {
  const headers = await getAuthHeaders();
  const r = await fetch(`${API}/api/v1/workshop/${workshopItemId}/promote`, {
    method: 'POST',
    headers: { ...headers, 'Content-Type': 'application/json' },
    body: JSON.stringify(opts),
  });
  if (!r.ok) throw new Error(`promote failed: ${r.status}`);
  return r.json();
}

export async function softDelete(workshopItemId: string): Promise<void> {
  const headers = await getAuthHeaders();
  const r = await fetch(`${API}/api/v1/workshop/${workshopItemId}`, {
    method: 'DELETE', headers,
  });
  if (!r.ok) throw new Error(`delete failed: ${r.status}`);
}
```

```typescript
// frontend/hooks/useWorkshopItems.ts
import { useEffect, useState } from 'react';
import { listByTeam, type WorkshopItem } from '../services/workshopService';

export function useWorkshopItems(teamId: string | null) {
  const [items, setItems] = useState<WorkshopItem[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    if (!teamId) { setItems([]); return; }
    let cancelled = false;
    setLoading(true);
    listByTeam(teamId)
      .then(r => { if (!cancelled) setItems(r.results); })
      .catch(e => { if (!cancelled) setError(e); })
      .finally(() => { if (!cancelled) setLoading(false); });
    return () => { cancelled = true; };
  }, [teamId]);

  return { items, loading, error };
}
```
